{{- /* The list on /releases/: every published release, newest first, with its downloads grouped by platform, its checksums, and its notes folded away. It reads data/releases.json, which is GitHub's own list (GET /repos/syshlted/drivel/releases) written by the Pages workflow just before the build. That step, not this template, is what talks to GitHub, so the workflow's token never reaches Hugo -- where any template, the theme's included, could read it -- and a build makes no network request of its own. The file is gitignored: a committed copy would be a second, stale answer to "what has been released?". A production build without the file is an error, so a deploy that lost the step fails rather than publishing a page that says nothing was ever released. `hugo server` without it says how to fetch it, and carries on. Platforms are recognised from the file names scripts/dist.sh gives them, and anything that does not match is still listed, under "Other files", so a change of naming there costs tidiness here and never a missing download. Styling is plain HTML plus drivel- classes from assets/css/custom.css; see the rule about hx: classes in docs/dev/website.md. @example {{< releases >}} */ -}} {{- $releases := index hugo.Data "releases" -}} {{- if not (reflect.IsSlice $releases) -}} {{- if hugo.IsProduction -}} {{- errorf "releases: data/releases.json is missing or not a list; the Pages workflow writes it before building" -}} {{- end -}} {{- warnf "releases: no data/releases.json; to preview the list, run from site/: gh api 'repos/syshlted/drivel/releases?per_page=100' > data/releases.json" -}} {{- $releases = slice -}} {{- end -}} {{- $published := where $releases "draft" false -}} {{- /* GitHub lists newest first; "Latest" is its own meaning of the word, the newest release that is not a pre-release. */ -}} {{- $latest := "" -}} {{- with where $published "prerelease" false -}} {{- $latest = (index . 0).tag_name -}} {{- end -}} {{- $osNames := dict "linux" "Linux" "freebsd" "FreeBSD" "darwin" "macOS" -}} {{- $archNames := dict "amd64" "x86-64" "arm64" "ARM64" -}} {{- $rpmArches := dict "amd64" "x86_64" "arm64" "aarch64" -}} {{- $tarRE := `^drivel-.+-([a-z0-9]+)-([a-z0-9]+)\.tar\.gz$` -}} {{- if not $published }}
